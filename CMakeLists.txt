cmake_minimum_required(VERSION 3.22.1)
project(aoe1_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/SDL/CMakeLists.txt")
  message(FATAL_ERROR "SDL2 missing. Run scripts/bootstrap-sdl.sh first.")
endif()

set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/SDL EXCLUDE_FROM_ALL)

if(ANDROID)
  add_library(main SHARED
    engine/src/main.cpp
    engine/src/platform/sdl_app.cpp
    engine/src/render/gl_renderer.cpp
    engine/src/input/touch_controller.cpp)
  target_include_directories(main PRIVATE engine/include)
  target_compile_definitions(main PRIVATE SDL_MAIN_HANDLED GL_GLEXT_PROTOTYPES=1)
  target_link_libraries(main PRIVATE SDL2 GLESv3 android log)
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_options(main PRIVATE -O3 -ffunction-sections -fdata-sections)
    target_link_options(main PRIVATE -Wl,--gc-sections)
  endif()
else()
  # Host CI compiles the platform-independent input module; GLES/Android
  # application targets are intentionally only generated by the Android NDK.
  add_library(aoe_host_check STATIC engine/src/input/touch_controller.cpp)
  target_include_directories(aoe_host_check PRIVATE engine/include)
  target_link_libraries(aoe_host_check PRIVATE SDL2)
endif()
